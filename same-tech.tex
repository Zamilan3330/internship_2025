\section{FAST, FIX протоколтой ижил технологиуд}
\subsection{FIXP + SBE (FIX Performance Session Layer + Simple Binary Encoding)}

\textbf{Тодорхойлолт}
 FIX Trading Community-гийн санал болгодог өндөр гүйцэтгэлтэй стек. FIXP нь сешн/холболтын давхаргыг, SBE нь мессежийн бинар кодчиллыг хангана.

\textbf{Зорилго:} FIX-ийн уян хатан байдал, стандартжилтыг хадгалж, FAST-тай харьцуулахад илүү бага latency, бага overhead-тай болгох.

\textbf{Техникийн онцлог}
\begin{itemize}
    \item \textbf{SBE:} урьдчилан тодорхойлсон схем дээр тулгуурласан компакт бинар кодчилол, CPU/кэшд ээлтэй
    \item \textbf{Дамжуулалт:} ихэвчлэн TCP/UDP, multicast-тэй хослуулж болно
\end{itemize}


{\textbf{Давуу тал:}}
\begin{itemize}
    \item Бага latency
    \item Өндөр throughput
    \item FIX экосистемтэй нийцтэй
\end{itemize}

{\textbf{Сул тал:}}
\begin{itemize}
    \item Схем удирдлага шаарддаг
    \item Хэрэгжүүлэлт, тестийн зардал FIX ASCII-гаас өндөр
\end{itemize}

\textbf{Хэзээ сонгох вэ:} ultra/low-latency арилжаа, биржийн market data/ордерийн систем, FAST-аас шилжихдээ. \par

\textbf{Nasdaq ITCH/OUCH (+ SoupBinTCP, MoldUDP64)}
    
\textbf{Тодорхойлолт}
\begin{itemize}
    \item \textbf{ITCH:} Nasdaq-ийн real-time market data (ордерийн ном, гүйлгээ) дамжуулах бинар протокол
    \item \textbf{OUCH:} ордер илгээх/засах/цуцлахад зориулсан order entry протокол
    \item \textbf{SoupBinTCP/MoldUDP64:} message framing, sequencing, retransmission-д ашиглагддаг тээврийн давхарга
\end{itemize}

\textbf{Зорилго:} өндөр хурд, бага latency-тай, бирж-шууд (direct) холболтод зориулсан proprietary стек.

\textbf{Техникийн онцлог}
\begin{itemize}
    \item \textbf{ITCH:} UDP multicast (MoldUDP64), sequence+replay механизмаар packet алдагдлыг нөхнө
    \item \textbf{OUCH:} TCP (SoupBinTCP), найдвартай, дараалал барина
\end{itemize}

{\textbf{Давуу тал:}}
\begin{itemize}
    \item Маш өндөр гүйцэтгэл
    \item Биржийн фийдтэй хамгийн ойр
    \item Стандартжсан үйл явдлын загвар
\end{itemize}

{\textbf{Сул тал:}}
\begin{itemize}
    \item Proprietary
    \item Бирж бүр өөр хувилбар/талбаруудтай
    \item Портлох зардал өндөр
\end{itemize}

\textbf{Хэзээ сонгох вэ:} Nasdaq direct market access (DMA), HFT, real-time data consumer.

\subsection{Cboe PITCH/BOE (PITCH market data, BOE order entry)}

\textbf{Тодорхойлолт}
\begin{itemize}
    \item \textbf{PITCH:} Cboe-ийн бинар market data протокол (ордерийн номын өөрчлөлт, гүйлгээ)
    \item \textbf{BOE/BOE+:} (Binary Order Entry) Cboe-ийн order entry протокол
\end{itemize}

\textbf{Зорилго:} Cboe-гийн платформ дээрх өндөр гүйцэтгэлийн дата түгээлт, ордерийн урсгал.

\textbf{Техникийн онцлог}
\begin{itemize}
    \item \textbf{PITCH:} ихэвчлэн UDP multicast + retransmission channel; sequence, snapshot+incremental стратеги
    \item \textbf{BOE:} TCP; бага latency-д чиглэсэн бинар бүтэц
\end{itemize}


{\textbf{Давуу тал:}}
\begin{itemize}
    \item Cboe экосистемд оновчтой
    \item Өндөр throughput
    \item Бага latency
\end{itemize}

{\textbf{Сул тал:}}
\begin{itemize}
    \item Proprietary
    \item Бирж хооронд протокол өөр тул олон биржийн дэмжлэг хэрэгтэй системд интеграцын ачаалал нэмэгдэнэ
\end{itemize}

\textbf{Хэзээ сонгох вэ:} Cboe-д DMA/market data хэрэглээ, HFT.

\subsection{Товч харьцуулалт}

\textbf{FIX ASCII vs FAST vs SBE/FIXP}
\begin{itemize}
    \item \textbf{FIX ASCII:} хүн ойлгоход амар, интеграц амар, latency/size их
    \item \textbf{FAST:} FIX-ийг шахаж дамжуулах стандарт; варьдаг талбар, темплэйт ашиглан bandwidth бууруулна
    \item \textbf{SBE/FIXP:} орчин үеийн бинар кодчилол, бага latency, өндөр throughput; FAST-ыг орлох чиг хандлагатай
\end{itemize}

\textbf{Nasdaq ITCH/OUCH, Cboe PITCH/BOE:}
\begin{itemize}
    \item \textbf{Market data:} ITCH/PITCH (UDP multicast, бинар, sequence/replay)
    \item \textbf{Order entry:} OUCH/BOE (TCP, бинар)
    \item \textbf{Давуу тал:} гүйцэтгэл
    \item \textbf{Сул тал:} бирж бүр тусдаа спецификтэй
\end{itemize}
